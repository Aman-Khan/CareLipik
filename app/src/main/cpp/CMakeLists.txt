cmake_minimum_required(VERSION 3.22.1)
project(carelipik_qwen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# The app's debug build still needs optimized matrix kernels for on-device inference.
add_compile_options(-O3)
add_library(carelipik_qwen SHARED qwen_jni.cpp)
target_link_options(carelipik_qwen PRIVATE "-Wl,-z,max-page-size=16384")
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    # Link-only stub: excluded from the APK. Android resolves the real public vendor driver.
    add_library(OpenCL SHARED opencl_link_stub.c)
    set(OpenCL_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tmp/l/OpenCL-Headers-2025.07.22" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY OpenCL CACHE STRING "" FORCE)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../tmp/l/llama.cpp-b11489" "llama")
    set_target_properties(ggml-cpu ggml-opencl PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
    foreach(native_target llama ggml ggml-base ggml-cpu ggml-opencl OpenCL)
        target_link_options(${native_target} PRIVATE "-Wl,-z,max-page-size=16384")
    endforeach()
    target_compile_definitions(carelipik_qwen PRIVATE CARELIPIK_ARM64)
    target_link_libraries(carelipik_qwen PRIVATE llama android log)
else()
    # Preserve existing ASR ABI support; Qwen gracefully falls back on other architectures.
    target_link_libraries(carelipik_qwen PRIVATE log)
endif()
